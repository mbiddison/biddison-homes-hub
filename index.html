const NEW_HOMES_FOLDER_ID = "0B5LaSP6hH0gNTVFNMGpNMWNJMVE"; const REMODELS_FOLDER_ID = "0B5LaSP6hH0gNV2hsazRadUlJSE0"; const MASTER_SITE_PHOTOS_DRIVE_ID = "1fqLRh4pUmQDPeLBZXl9k93rqLywYMz8a"; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Dynamic Address Addition if (data.action === "add_address") { let sheet = ss.getSheetByName("Addresses"); if (!sheet) sheet = ss.insertSheet("Addresses"); sheet.appendRow(["", data.value, new Date()]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // 2. Dynamic Trade Addition if (data.action === "add_trade") { let sheet = ss.getSheetByName("Trades"); if (!sheet) sheet = ss.insertSheet("Trades"); sheet.appendRow(["", data.value, new Date()]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } // 3. Routing Submissions if (data.type === "site_photo_catalog") { return handleSitePhotoUpload(data); } else { return handleWorkOrderSubmission(data); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const param = e.parameter.endpoint; // 1. Master Sheet Addresses and Trades Endpoint if (param === "get_lists") { let addrSheet = ss.getSheetByName("Addresses"); let addresses = []; if (addrSheet && addrSheet.getLastRow() > 0) { let rawAddrData = addrSheet.getRange(1, 1, addrSheet.getLastRow(), 2).getValues(); for (let i = 0; i < rawAddrData.length; i++) { let status = String(rawAddrData[i][0]).trim().toUpperCase(); let address = String(rawAddrData[i][1]).trim(); if (address && status !== "X") { addresses.push(address); } } } let tradeSheet = ss.getSheetByName("Trades"); let trades = []; if (tradeSheet && tradeSheet.getLastRow() > 0) { let rawTradeData = tradeSheet.getRange(1, 1, tradeSheet.getLastRow(), 2).getValues(); for (let j = 0; j < rawTradeData.length; j++) { let status = String(rawTradeData[j][0]).trim().toUpperCase(); let tradeName = String(rawTradeData[j][1]).trim(); if (tradeName && status !== "X") { trades.push(tradeName); } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", addresses: addresses, trades: trades })).setMimeType(ContentService.MimeType.JSON); } // 2. Job Info Category Lookup Endpoint (Fetch top-level active job folders) if (param === "get_job_info_categories") { const cat = e.parameter.category; let targetId = (cat === "remodels") ? REMODELS_FOLDER_ID : NEW_HOMES_FOLDER_ID; let parentFolder = DriveApp.getFolderById(targetId); let subfolders = parentFolder.getFolders(); let jobs = []; while (subfolders.hasNext()) { let f = subfolders.next(); let fName = f.getName(); // Exclude top-level job folders if marked with office if (!isOfficeProtected(fName)) { jobs.push({ name: fName, id: f.getId() }); } } jobs.sort((a, b) => a.name.localeCompare(b.name)); return ContentService.createTextOutput(JSON.stringify({ status: "success", jobs: jobs })).setMimeType(ContentService.MimeType.JSON); } // 3. Job Info Subfolder & File Tree Lookup Endpoint (Filtered by office protection) if (param === "get_job_files") { const folderId = e.parameter.folderId; if (!folderId) return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No folder ID provided" })).setMimeType(ContentService.MimeType.JSON); let folder = DriveApp.getFolderById(folderId); let items = fetchFolderContentsRecursively(folder); return ContentService.createTextOutput(JSON.stringify({ status: "success", folderUrl: folder.getUrl(), items: items })).setMimeType(ContentService.MimeType.JSON); } // 4. Tab 2 Photo Folder Lookup Endpoint const address = e.parameter.address; if (address) { return handleFolderLookup(address); } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid endpoint" })).setMimeType(ContentService.MimeType.JSON); } // Helper function to detect if a file or folder name should be hidden function isOfficeProtected(name) { if (!name) return false; let lower = name.trim().toLowerCase(); // Catches "[office]", "office", "_office", "(office)", or anything containing "office" return lower.includes("office"); } // Helper function to recursively search subfolders and files while filtering out [Office] items function fetchFolderContentsRecursively(folder) { let items = []; // 1. Process Subfolders let childFolders = folder.getFolders(); while (childFolders.hasNext()) { let cf = childFolders.next(); let cfName = cf.getName(); // If the subfolder is NOT office protected, include it if (!isOfficeProtected(cfName)) { items.push({ name: cfName, url: cf.getUrl(), type: "folder" }); } } // 2. Process Files let childFiles = folder.getFiles(); while (childFiles.hasNext()) { let file = childFiles.next(); let fileName = file.getName(); // If the file is NOT office protected, include it if (!isOfficeProtected(fileName)) { items.push({ name: fileName, url: file.getUrl(), type: "file" }); } } return items; } function handleFolderLookup(address) { let masterFolder = DriveApp.getFolderById(MASTER_SITE_PHOTOS_DRIVE_ID); let subfolders = masterFolder.getFoldersByName(address); let folder; if (subfolders.hasNext()) { folder = subfolders.next(); } else { folder = masterFolder.createFolder(address); } return ContentService.createTextOutput(JSON.stringify({ status: "success", folderUrl: folder.getUrl() })).setMimeType(ContentService.MimeType.JSON); } function handleWorkOrderSubmission(data) { let masterFolder = DriveApp.getFolderById(MASTER_SITE_PHOTOS_DRIVE_ID); let subfolders = masterFolder.getFoldersByName(data.address); let folder = subfolders.hasNext() ? subfolders.next() : masterFolder.createFolder(data.address); let photoUrls = []; if (data.photos && data.photos.length > 0) { data.photos.forEach((p, idx) => { let blob = Utilities.newBlob(Utilities.base64Decode(p.base64), p.mimeType, `WO_${data.woNumber}_${idx+1}_${p.name}`); let file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoUrls.push(file.getUrl()); }); } let woFileName = `${data.woNumber} - ${data.tradePartner}.txt`; let woContent = `BIDDISON HOMES WORK ORDER: ${data.woNumber}\nDate: ${data.creationDate}\nAddress: ${data.address}\nTrade Partner: ${data.tradePartner}\nSupervisor: ${data.supervisor}\n\nWork Items:\n${data.workItems}\n\nSpecial Instructions:\n${data.specialInstructions}`; folder.createFile(woFileName, woContent); if (data.completionDate) { let targetDate = new Date(data.completionDate); CalendarApp.createAllDayEvent(`WO Due: ${data.woNumber} - ${data.tradePartner} (${data.address})`, targetDate); } return ContentService.createTextOutput(JSON.stringify({ status: "success", photoUrls: photoUrls })).setMimeType(ContentService.MimeType.JSON); } function handleSitePhotoUpload(data) { let masterFolder = DriveApp.getFolderById(MASTER_SITE_PHOTOS_DRIVE_ID); let subfolders = masterFolder.getFoldersByName(data.address); let folder = subfolders.hasNext() ? subfolders.next() : masterFolder.createFolder(data.address); let photoUrls = []; if (data.photos && data.photos.length > 0) { data.photos.forEach((p, idx) => { let blob = Utilities.newBlob(Utilities.base64Decode(p.base64), p.mimeType, `SitePhoto_${idx+1}_${p.name}`); let file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoUrls.push(file.getUrl()); }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", folderUrl: folder.getUrl(), photoUrls: photoUrls })).setMimeType(ContentService.MimeType.JSON); }
